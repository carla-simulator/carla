cmake_minimum_required(VERSION 3.28)
project(CarlaRawDataBufferTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Embed)
find_package(GTest REQUIRED)
add_executable(test_raw_data_buffer test_raw_data_buffer.cpp)
target_include_directories(test_raw_data_buffer PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../carla/src")
target_link_libraries(test_raw_data_buffer PRIVATE GTest::gtest_main Python3::Python)
if(MSVC)
  target_compile_options(test_raw_data_buffer PRIVATE /W4)
else()
  target_compile_options(test_raw_data_buffer PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME raw_data_buffer COMMAND test_raw_data_buffer)
